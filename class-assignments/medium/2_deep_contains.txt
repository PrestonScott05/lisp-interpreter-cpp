; Problem 2: Deep contains (class list #3)
;
; Given a list containing numbers and nested lists, determine if the list
; contains the number x. The list may contain nested lists to any depth. You may
; assume all values are lists or numbers.


(def deep_contains (lst x)
    (cond (
        (nil? lst) ()
        (not? (list? lst)) (eq? lst x)
        'T (
            or? 
                (deep_contains (car lst) x)
                (deep_contains (cdr lst) x)
        )
    ))
)
