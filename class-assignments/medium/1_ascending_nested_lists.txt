; Problem 1: Nested ascending (class list #2)
;
; Given a list of number lists eg: '((1 2 3) (9) (4 5)), determine if all the
; nested lists are sorted (ascending). Return false if any sub list is out of
; order.

(def check_ascending (inlist)
    (cond (
        (nil? inlist) 'T
        (not? (list? inlist)) ()
        (nil? (cdr inlist)) 'T
        (gt? (car inlist) (car (cdr inlist))) ()
        'T (check_ascending (cdr inlist))
    ))
)

(def process_nested_lists (nested_list)
    (cond (
        (nil? nested_list) 'T
        (not? (list? nested_list)) ()
        (nil? (check_ascending (car nested_list))) ()
        'T (process_nested_lists (cdr nested_list))
    ))
)

(def gt? (num1 num2)
    (lt num2 num1)
