; Given a list (flat) of numbers the function returns nil if the list is not in
; ascending sorted order.

(def check_ascending (inlist)
    (cond (
        (nil? inlist) 'T
        (not? (list? inlist)) ()
        (nil? (cdr inlist)) 'T
        (gt? (car inlist) (car (cdr inlist))) ()
        'T (check_ascending (cdr inlist))
    ))
)

(def gt? (num1 num2)
    (lt num2 num1)
