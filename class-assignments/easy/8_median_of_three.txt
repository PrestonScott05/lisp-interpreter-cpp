; Problem 8: Median of three (class list #4)
;
; Given three numbers return the median.

(def median (a b c)
    (cond (
        (or?
            (and?
                (le? a b)
                (le? b c)
            )
            (and?
                (le? c b)
                (le? b a)
            )
        ) b
        (or? 
            (and? 
                (le? b a)
                (le? a c)
            )
            (and? 
                (le? c a)
                (le? a b)
            )
        ) a
        (or?
            (and? 
                (le? a c)
                (le? c b)
            )
            (and?
                (le? b c)
                (le? c a)
            )
        ) c
    ))
)

(def le? (a b)
    (or? 
        (lt a b)
        (eq? a b)
    )
)
